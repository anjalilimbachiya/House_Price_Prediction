def get_dataset():
    try:
        df = load_data(DATA_PATH)
        return df
    except Exception:
        return pd.DataFrame()


@st.cache_data
def get_metrics():
    if not METRICS_PATH.exists():
        return pd.DataFrame()
    try:
        return pd.read_csv(METRICS_PATH)
    except Exception:
        return pd.DataFrame()


@get_metrics.cache_data
def cache_metrics():
    return get_metrics()


def model_exists():
    return MODEL_PATH.exists()


def format_currency(value):
    if pd.isna(value):
        return "$0"
    return f"${float(value):,.0f}"


def load_model_if_exists():
    if not MODEL_PATH.exists():
        return None
    try:
        import joblib
        return joblib.load(MODEL_PATH)
    except Exception:
        return None

def render_prediction_page(df):
    render_header("AI PROPERTY VALUATION", "Enter property details and let our AI estimate its market value.")

    model = load_model_if_exists()
    if model is None:
        st.warning("⚠️ Model not trained yet. Please run: python train.py")
        st.code("python train.py")
        return

    locations = sorted(df["Location"].dropna().unique().tolist()) if not df.empty else ["Suburban"]
    conditions = sorted(df["Condition"].dropna().unique().tolist()) if not df.empty else ["Good"]

    with st.container():
        left, right = st.columns([1.3, 0.9])

        with left:
            st.markdown("<div class='panel-title'>PROPERTY DETAILS</div>", unsafe_allow_html=True)
            with st.form("valuation_form", clear_on_submit=False):
                c1, c2 = st.columns(2)
                with c1:
                    area = st.number_input("Area", min_value=100.0, value=2200.0, step=25.0, key="area_in")
                    bedrooms = st.number_input("Bedrooms", min_value=1, value=3, step=1, key="bed_in")
                    bathrooms = st.number_input("Bathrooms", min_value=1, value=2, step=1, key="bath_in")
                    floors = st.number_input("Floors", min_value=1, value=2, step=1, key="floors_in")
                with c2:
                    year_built = st.number_input("Year Built", min_value=1900, max_value=2100, value=2012, step=1, key="year_in")
                    location = st.selectbox("Location", locations, key="loc_in")
                    condition = st.selectbox("Condition", conditions, key="cond_in")
                    garage = st.selectbox("Garage", ["Yes", "No"], key="garage_in")

                submitted = st.form_submit_button("✨ ESTIMATE PROPERTY VALUE", use_container_width=True)

        with right:
            st.markdown("<div class='panel-title'>LIVE PROPERTY SUMMARY</div>", unsafe_allow_html=True)
            st.markdown(
                f"""
                <div class='summary-stack'>
                    <div class='summary-card'><span>AREA</span><strong>{int(area):,} sq ft</strong></div>
                    <div class='summary-card'><span>BEDROOMS</span><strong>{int(bedrooms)}</strong></div>

                "Condition": condition,
                "Garage": garage,

                <div class='result-header'>AI ESTIMATED VALUE</div>

                <div class='result-subtitle'>Estimated Property Value</div>

def render_market_analytics_page(df):
    render_header("MARKET ANALYTICS", "Property intelligence and market signals.")
    if df.empty:
        st.warning("No dataset available.")
        return

    tabs = st.tabs(["Overview", "Location", "Property Features", "Price Analysis"])

    with tabs[0]:
        c1, c2, c3, c4 = st.columns(4)
        with c1:
            render_metric_card("💰", "Average Price", format_currency(df["Price"].mean()), "Mean property value")
        with c2:
            render_metric_card("📊", "Median Price", format_currency(df["Price"].median()), "Middle market value")
        with c3:
            render_metric_card("🏠", "Price/Area", f"${(df['Price'] / df['Area']).mean():,.2f}", "Per square foot")
        with c4:
            render_metric_card("📍", "Top Location", df.groupby('Location')['Price'].mean().idxmax(), "Highest average price")

        fig = px.histogram(df, x="Price", nbins=35, title="Property Price Distribution", template="plotly_dark")
        fig.update_traces(marker_color="#7C3AED")
        render_chart_card(fig)

    with tabs[1]:
        loc_means = df.groupby("Location", as_index=False)["Price"].mean().sort_values("Price", ascending=False)
        fig = px.bar(loc_means, x="Location", y="Price", title="Average Price by Location", template="plotly_dark", color_discrete_sequence=["#06B6D4", "#7C3AED", "#EC4899"])
        fig.update_layout(showlegend=False)
        render_chart_card(fig)

    with tabs[2]:
        bedroom_summary = df.groupby("Bedrooms", as_index=False)["Price"].mean()
        fig = px.line(bedroom_summary, x="Bedrooms", y="Price", title="Bedrooms vs Average Price", template="plotly_dark")
        fig.update_traces(line_color="#06B6D4", line_shape="spline")
        render_chart_card(fig)

        cond_summary = df.groupby("Condition", as_index=False)["Price"].mean().sort_values("Price", ascending=False)
        fig2 = px.bar(cond_summary, x="Condition", y="Price", title="Condition vs Average Price", template="plotly_dark", color_discrete_sequence=["#EC4899", "#7C3AED", "#06B6D4", "#22C55E"])
        render_chart_card(fig2)

    with tabs[3]:
        fig = px.scatter(df, x="Area", y="Price", title="Area vs Price", template="plotly_dark", hover_data=["Bedrooms", "Bathrooms", "Location"])
        fig.update_traces(marker=dict(size=8, line=dict(width=0.5, color="#F8FAFC")))
        render_chart_card(fig)


def render_model_intelligence_page():
    render_header("MODEL INTELLIGENCE", "Performance comparison and prediction quality.")
    metrics_df = get_metrics()
    if metrics_df.empty:
        st.warning("⚠️ No model metrics available. Please run python train.py first.")
        st.code("python train.py")
        return

    metrics_df = metrics_df.sort_values("R2", ascending=False).reset_index(drop=True)
    best_row = metrics_df.iloc[0]

    cols = st.columns(4)
    for i, row in enumerate(metrics_df.itertuples(index=False)):
        with cols[i]:
            accent = "#22C55E" if i == 0 else "#7C3AED"
            st.markdown(
                f"""
                <div class='model-card' style='border-color:{accent};'>
                    <div class='model-name'>{row.Model}</div>
                    <div class='model-metric'>R²: {float(row.R2):.4f}</div>
                    <div class='model-metric'>RMSE: {float(row.RMSE):.4f}</div>
                    <div class='model-metric'>MAE: {float(row.MAE):.4f}</div>
                    {"<div class='best-badge'>🏆 BEST MODEL</div>" if i == 0 else ""}
                </div>
                """,
                unsafe_allow_html=True,
            )

    fig = px.bar(metrics_df, x="Model", y="R2", title="Model Performance Comparison", template="plotly_dark", color="Model")
    fig.update_layout(showlegend=False)
    render_chart_card(fig)

    model = load_model_if_exists()
    if model is not None:
        data = get_dataset()
        if not data.empty:
            predictions = model.predict(data[FEATURE_COLUMNS])
            compare_df = pd.DataFrame({"Actual Price": data["Price"], "Predicted Price": predictions})
            fig2 = px.scatter(compare_df, x="Actual Price", y="Predicted Price", title="Actual vs Predicted Price", template="plotly_dark")
            min_v = min(compare_df["Actual Price"].min(), compare_df["Predicted Price"].min())
            max_v = max(compare_df["Actual Price"].max(), compare_df["Predicted Price"].max())
            fig2.add_scatter(x=[min_v, max_v], y=[min_v, max_v], mode="lines", line=dict(color="#06B6D4", dash="solid"), name="Ideal prediction")
            render_chart_card(fig2)
